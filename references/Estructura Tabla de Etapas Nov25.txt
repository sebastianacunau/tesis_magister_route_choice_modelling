periodoSubida	character varying(50)	Periodo asociado a la hora de subida
periodoBajada	character varying(50)	Periodo asociado a la hora de bajada
tiempoIniExpedicion	character varying(50)	Tiempo de inicio de la expedición
contrato	integer	Códificación del tipo de contrato (Dic_tipo_contrato)
